const playAudioBeep = (freq = 880, type: OscillatorType = 'sine', duration = 0.08) => {
  try {
    const AudioCtx = window.AudioContext || (window as unknown as { webkitAudioContext: typeof AudioContext }).webkitAudioContext;
    if (!AudioCtx) return;
    const ctx = new AudioCtx();
    const osc = ctx.createOscillator();
    const gain = ctx.createGain();
    osc.type = type;
    osc.frequency.setValueAtTime(freq, ctx.currentTime);
    gain.gain.setValueAtTime(0.04, ctx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.0001, ctx.currentTime + duration);
    osc.connect(gain);
    gain.connect(ctx.destination);
    osc.start();
    osc.stop(ctx.currentTime + duration);
  } catch {
    // Ignore audio context autoplay errors
  }
};

  const [soundEnabled, setSoundEnabled] = useState<boolean>(true);
  const [showManifestModal, setShowManifestModal] = useState<boolean>(false);

  const [copiedHash, setCopiedHash] = useState<string | null>(null);

    if (isPlaying) {
      timer = setInterval(() => {
        setActivePhase((prev) => {
          const next = prev >= 12 ? 1 : prev + 1;
          setExpandedPhase(next);
          if (soundEnabled) {
            playAudioBeep(600 + next * 40, 'triangle', 0.06);
          }
          return next;
        });
      }, playbackSpeed);
    }
    return () => clearInterval(timer);
  }, [isPlaying, playbackSpeed, soundEnabled]);

    const matchesSearch =
      phase.name.toLowerCase().includes(searchQuery.toLowerCase()) ||
      phase.thName.toLowerCase().includes(searchQuery.toLowerCase()) ||
      phase.keyAction.toLowerCase().includes(searchQuery.toLowerCase()) ||
      phase.legalBinding.toLowerCase().includes(searchQuery.toLowerCase()) ||
      phase.code.toLowerCase().includes(searchQuery.toLowerCase());

    const matchesExhibit =
      selectedExhibitFilter === 'ALL' || phase.exhibitId === selectedExhibitFilter;

    return matchesSearch && matchesExhibit;
  });

  const handleCopy = (text: string, label: string) => {
    navigator.clipboard.writeText(text);
    setCopiedHash(label);
    setTimeout(() => setCopiedHash(null), 2000);
  };

  const generateManifestJSON = () => {
    return JSON.stringify(
      {
        system_attestation: 'ZYRQUEN-SLA-TH-ETA-v1.0',
        runtime_state: 'LOCKED_FROZEN_v1.2_LTS',
        sovereign_identity: '#EP-SOVEREIGN-01 OMEGA-1 SUPREME',
        total_sla_latency_ms: totalLatency.toFixed(2),
        ssot_baseline_drift: '0.00%',
        genesis_merkle_root: '909ab814479844d8a14816bed34cdbb07528e18501da86fc4691763a43fa4c68',
        hsms_quorum: '10/10 REAL_HSM',
        golden_seals_count: 14902,
        phases: OMEGA_PHASES.map((p) => ({
          phase_code: p.code,
          phase_name: p.name,
          latency_ms: p.latencyMs,
          legal_binding: p.legalBinding,
          exhibit_id: p.exhibitId,
          hash: p.hashSignature
        }))
      },
      null,
      2
    );
  };

  const generateManifestCSV = () => {
    const headers = 'PhaseCode,PhaseName,LatencyMs,ExhibitID,LegalBinding,HashSignature\n';
    const rows = OMEGA_PHASES.map(
      (p) =>
        `"${p.code}","${p.name}",${p.latencyMs},"${p.exhibitId}","${p.legalBinding}","${p.hashSignature}"`
    ).join('\n');
    return headers + rows;
  };

  return (
    <div className="w-full min-h-screen bg-slate-950 text-slate-100 font-sans p-4 md:p-8 space-y-8 selection:bg-cyan-500 selection:text-slate-950">
      
      {}
      <header className="relative border border-slate-800 rounded-2xl bg-slate-900/80 backdrop-blur-xl p-6 shadow-2xl overflow-hidden">
        <div className="absolute -right-20 -top-20 w-80 h-80 bg-cyan-500/10 rounded-full blur-3xl pointer-events-none" />
        <div className="absolute -left-20 -bottom-20 w-80 h-80 bg-violet-500/10 rounded-full blur-3xl pointer-events-none" />

        <div className="flex flex-col lg:flex-row lg:items-center justify-between gap-6 relative z-10">
          <div className="space-y-2">
            <div className="flex flex-wrap items-center gap-2 text-xs font-mono">
              <span className="px-2.5 py-1 rounded-md bg-cyan-950 border border-cyan-500/40 text-cyan-400 font-semibold flex items-center gap-1.5">
                <span className="w-2 h-2 rounded-full bg-cyan-400 animate-pulse" />
                ZYRQUEN-SLA-TH-ETA-v1.0
              </span>
              <span className="px-2.5 py-1 rounded-md bg-emerald-950 border border-emerald-500/40 text-emerald-400 font-semibold flex items-center gap-1.5">
                <Lock className="w-3.5 h-3.5" />
                LOCKED_FROZEN_v1.2_LTS
              </span>
              <span className="px-2.5 py-1 rounded-md bg-amber-950 border border-amber-500/40 text-amber-300 font-semibold">
                SSoT Δ0 (Drift 0.00%)
              </span>
            </div>

            <h1 className="text-2xl md:text-4xl font-extrabold tracking-tight text-white flex items-center gap-3">
              <Scale className="w-8 h-8 text-amber-400 shrink-0" />
              12-Phase Omega Sequence Visualizer
            </h1>
            <p className="text-slate-400 text-sm md:text-base max-w-3xl">
              รายงานและผังลำดับกระบวนการประทับตราสัจจะพยานหลักฐานดิจิทัล (Court Evidence Exhibits จพ.01 – จพ.07) 
              ด้วยระบบมติสภา 10/10 REAL_HSM และลายมือชื่อต้านควอนตัม (Post-Quantum Cryptography)

          <div className="flex flex-wrap items-center gap-3 shrink-0">
            <button
              onClick={() => setShowManifestModal(true)}
              className="px-4 py-2.5 rounded-xl bg-gradient-to-r from-cyan-600 to-blue-600 hover:from-cyan-500 hover:to-blue-500 text-white text-sm font-semibold shadow-lg shadow-cyan-900/30 flex items-center gap-2 transition-all active:scale-95"
            >
              <Download className="w-4 h-4" />
              Export Court Manifest (JSON/CSV)
            </button>
          </div>
        </div>

        {}
        <div className="grid grid-cols-2 md:grid-cols-4 gap-4 mt-6 pt-6 border-t border-slate-800">
          <div className="bg-slate-950/60 p-3.5 rounded-xl border border-slate-800/80">
            <div className="text-xs text-slate-400 flex items-center gap-1.5 mb-1 font-mono">
              <Clock className="w-3.5 h-3.5 text-cyan-400" />
              TOTAL SLA LATENCY
            </div>
            <div className="text-xl md:text-2xl font-bold font-mono text-cyan-400">
              {totalLatency.toFixed(2)} <span className="text-xs text-slate-400">ms</span>
            </div>
            <div className="text-[11px] text-slate-500 mt-1">SLA Limit &lt; 142.00 ms (Passed 4x)</div>
          </div>

          <div className="bg-slate-950/60 p-3.5 rounded-xl border border-slate-800/80">
            <div className="text-xs text-slate-400 flex items-center gap-1.5 mb-1 font-mono">
              <ShieldCheck className="w-3.5 h-3.5 text-emerald-400" />
              DECA_HSM QUORUM
            </div>
            <div className="text-xl md:text-2xl font-bold font-mono text-emerald-400">
              10 / 10 <span className="text-xs text-slate-400">REAL_HSM</span>
            </div>
            <div className="text-[11px] text-slate-500 mt-1">Unanimous Council Consensus</div>
          </div>

          <div className="bg-slate-950/60 p-3.5 rounded-xl border border-slate-800/80">
            <div className="text-xs text-slate-400 flex items-center gap-1.5 mb-1 font-mono">

            >
              <option value="ALL">All Exhibits (จพ.01 - จพ.07)</option>
              <option value="จพ.01">จพ.01: Genesis & Root Hash</option>
              <option value="จพ.02">จพ.02: Hardware TSA Token</option>
              <option value="จพ.03">จพ.03: 10/10 HSM Quorum</option>

            <input
              type="text"
              placeholder="Search phase, legal act, hash..."
              value={searchQuery}
              onChange={(e) => setSearchQuery(e.target.value)}
              className="w-full pl-9 pr-3 py-2 bg-slate-950 border border-slate-800 rounded-xl text-xs font-sans text-slate-200 placeholder-slate-500 focus:outline-none focus:border-cyan-500"
            />
          </div>
        </div>

        {/* Vertical Glowing Connector Line */}

        <div className="space-y-8">
          {filteredPhases.map((phase, index) => {
            const isActive = activePhase === phase.id;
            const isExpanded = expandedPhase === phase.id;
            const isEven = index % 2 === 0;

              <motion.div
                key={phase.id}
                initial={{ opacity: 0, y: 30 }}
                whileInView={{ opacity: 1, y: 0 }}
                viewport={{ once: true, margin: '-50px' }}
                transition={{ duration: 0.5, delay: index * 0.05 }}
                className={`relative flex flex-col md:flex-row items-center gap-6 ${
                  isEven ? 'md:flex-row-reverse' : ''
                }`}
              >
                {/* Center Node Marker Icon */}

                  <motion.button
                    whileHover={{ scale: 1.2 }}
                    whileTap={{ scale: 0.9 }}
                    onClick={() => {
                      setActivePhase(phase.id);
                      setExpandedPhase(isExpanded ? null : phase.id);
                      if (soundEnabled) playAudioBeep(500 + phase.id * 30, 'sine', 0.05);
                    }}
                    className={`w-10 h-10 rounded-full border-2 flex items-center justify-center text-xs font-mono font-bold transition-all shadow-xl ${
                      isActive
                        ? 'bg-cyan-500 border-white text-slate-950 ring-4 ring-cyan-500/30 scale-110'
                        : 'bg-slate-900 border-slate-700 text-cyan-400 hover:border-cyan-400'
                    }`}
                  >
                    {phase.id}
                  </motion.button>
                </div>

                {/* Timeline Card Content */}
                <div className="w-full md:w-[calc(50%-2.5rem)] pl-12 md:pl-0">
                  <div
                    className={`border rounded-2xl p-5 backdrop-blur-xl transition-all cursor-pointer shadow-xl ${
                      isActive
                        ? 'bg-slate-900/90 border-cyan-500/60 ring-1 ring-cyan-500/40 shadow-cyan-950/50'
                        : 'bg-slate-900/50 border-slate-800/80 hover:border-slate-700 hover:bg-slate-900/80'
                    }`}
                    onClick={() => {
                      setActivePhase(phase.id);
                      setExpandedPhase(isExpanded ? null : phase.id);
                      if (soundEnabled) playAudioBeep(500 + phase.id * 30, 'sine', 0.05);
                    }}
                  >
                    {/* Phase Header Badges */}
                    <div className="flex items-center justify-between gap-2 mb-2">
                      <div className="flex items-center gap-2">
                        <span className="px-2 py-0.5 rounded-md bg-slate-950 border border-slate-800 text-cyan-400 font-mono text-xs font-semibold">
                          {phase.code}
                        </span>
                        <span className="px-2 py-0.5 rounded-md bg-amber-950/60 border border-amber-500/30 text-amber-300 font-mono text-[11px] font-bold">
                          {phase.exhibitId}
                        </span>

                    </div>

                      {phase.thKeyAction}

                        <ChevronUp className="w-4 h-4 text-cyan-400 shrink-0" />

                    <span className="inline-flex items-center gap-1.5 px-2 py-1 rounded bg-amber-950/40 border border-amber-500/20 text-amber-300 font-sans">
                      <Scale className="w-3 h-3 shrink-0" />
                      {ex.legalBasis}
                    </span>
                  </td>
                  <td className="p-3 font-mono text-cyan-300 text-[11px]">
                    {ex.techStandard}
                  </td>
                  <td className="p-3 text-right">
                    <span className="px-2 py-1 rounded-md bg-emerald-950 border border-emerald-500/40 text-emerald-400 font-mono text-[10px] font-bold">
                      {ex.status}
                    </span>
                  </td>
                </tr>
              ))}
            </tbody>
          </table>
        </div>
      </section>

      {}
      <AnimatePresence>
        {showManifestModal && (
          <motion.div
            initial={{ opacity: 0 }}
            animate={{ opacity: 1 }}
            exit={{ opacity: 0 }}
            className="fixed inset-0 z-50 bg-slate-950/80 backdrop-blur-md flex items-center justify-center p-4"
            onClick={() => setShowManifestModal(false)}
          >
            <motion.div
              initial={{ scale: 0.9, opacity: 0 }}
              animate={{ scale: 1, opacity: 1 }}
              exit={{ scale: 0.9, opacity: 0 }}
              className="bg-slate-900 border border-slate-800 rounded-2xl max-w-3xl w-full max-h-[85vh] flex flex-col shadow-2xl overflow-hidden"
              onClick={(e) => e.stopPropagation()}
            >
              {/* Modal Header */}
              <div className="p-5 border-b border-slate-800 flex items-center justify-between bg-slate-950">
                <div className="flex items-center gap-3">
                  <Terminal className="w-5 h-5 text-cyan-400" />
                  <div>
                    <h3 className="font-bold text-white text-base">
                      Court Evidence Manifest Dossier Export
                    </h3>
                    <p className="text-xs text-slate-400">
                      zyrquen-court-manifest (.json / .csv)
                    </p>
                  </div>
                </div>

                <div className="flex items-center gap-2">
                  <button
                    onClick={() => setManifestFormat('JSON')}
                    className={`px-3 py-1 rounded-lg text-xs font-mono font-semibold transition-colors ${
                      manifestFormat === 'JSON'
                        ? 'bg-cyan-500 text-slate-950'
                        : 'bg-slate-800 text-slate-400 hover:text-white'
                    }`}
                  >
                    JSON Format
                  </button>
                  <button
                    onClick={() => setManifestFormat('CSV')}
                    className={`px-3 py-1 rounded-lg text-xs font-mono font-semibold transition-colors ${
                      manifestFormat === 'CSV'
                        ? 'bg-cyan-500 text-slate-950'
                        : 'bg-slate-800 text-slate-400 hover:text-white'
                    }`}
                  >
                    CSV Format
                  </button>
                </div>
              </div>

              {/* Modal Code Viewer Body */}
              <div className="p-5 overflow-y-auto flex-1 font-mono text-xs bg-slate-950 text-cyan-300">
                <pre className="whitespace-pre-wrap break-all selection:bg-cyan-900">
                  {manifestFormat === 'JSON' ? generateManifestJSON() : generateManifestCSV()}
                </pre>
              </div>

              {/* Modal Footer Controls */}
              <div className="p-4 border-t border-slate-800 bg-slate-950 flex items-center justify-between">
                <span className="text-xs text-slate-400">
                  ISO/IEC 27037 & ETDA Certified Output
                </span>

                <div className="flex items-center gap-3">
                  <button
                    onClick={() =>
                      handleCopy(
                        manifestFormat === 'JSON' ? generateManifestJSON() : generateManifestCSV(),
                        'manifest'
                      )
                    }
                    className="px-4 py-2 rounded-xl bg-slate-800 hover:bg-slate-700 text-white text-xs font-semibold flex items-center gap-2 transition-colors"
                  >
                    {copiedHash === 'manifest' ? (
                      <Check className="w-4 h-4 text-emerald-400" />
                    ) : (
                      <Copy className="w-4 h-4" />
                    )}
                    {copiedHash === 'manifest' ? 'Copied to Clipboard' : 'Copy Manifest'}
                  </button>

                  <button
                    onClick={() => setShowManifestModal(false)}
                    className="px-4 py-2 rounded-xl bg-cyan-600 hover:bg-cyan-500 text-white text-xs font-semibold transition-colors"
                  >
                    Close
                  </button>
                </div>
              </div>
            </motion.div>
          </motion.div>
        )}
      </AnimatePresence>

      {}
      <footer className="text-center py-6 border-t border-slate-800/80 text-xs text-slate-500 space-y-1">
        <div>
          ZYRQUEN SLA Attestation Engine &bull; Sovereign Sovereign Control Unit #EP-SOVEREIGN-01
        </div>
        <div>
          Strict Enforcement Ω601–Ω1000 &bull; Baseline Drift 0.00% &bull; All System Rights Sealed
        </div>
      </footer>
    </div>
  );
};

export default CourtEvidenceTimeline;
